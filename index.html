PostgreSQL Database Migration Script

Copy and execute this script inside your Supabase SQL Editor to create tables, indexes, and Row Level Security (RLS) policies.

-- GOD OF AUTOMATION POSTGRESQL SCHEMA CREATE EXTENSION IF NOT EXISTS "uuid-ossp"; CREATE TABLE IF NOT EXISTS public.leads_and_inquiries ( id UUID PRIMARY KEY DEFAULT uuid_generate_v4(), reference_id VARCHAR(30) UNIQUE NOT NULL, created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL, full_name VARCHAR(255) NOT NULL, email VARCHAR(255) NOT NULL, phone VARCHAR(50), selected_service VARCHAR(255) DEFAULT 'General Consultation', subject VARCHAR(255), message TEXT, source_page VARCHAR(255) DEFAULT 'Homepage', status VARCHAR(50) DEFAULT 'New' ); CREATE TABLE IF NOT EXISTS public.consultation_bookings ( id UUID PRIMARY KEY DEFAULT uuid_generate_v4(), reference_id VARCHAR(30) UNIQUE NOT NULL, created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL, client_name VARCHAR(255) NOT NULL, email VARCHAR(255) NOT NULL, phone VARCHAR(50), target_deployment VARCHAR(255) NOT NULL, preferred_date DATE NOT NULL, preferred_time VARCHAR(50) NOT NULL, notes TEXT, status VARCHAR(50) DEFAULT 'Scheduled' );
